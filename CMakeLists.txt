cmake_minimum_required(VERSION 3.29)
project(noorrhi LANGUAGES C CXX)

include(cmake/CompilerOptions.cmake)
include(cmake/NoorRHIShaders.cmake)

option(NOORRHI_BUILD_TOOLS "Build the example tools" ${PROJECT_IS_TOP_LEVEL})
option(NOORRHI_BUILD_TESTS "Build the tests" ${PROJECT_IS_TOP_LEVEL})
if(EMSCRIPTEN)
    set(NOORRHI_BACKEND WebGpu CACHE STRING "Graphics API: Vulkan or WebGpu" FORCE)
else()
    set(NOORRHI_BACKEND Vulkan CACHE STRING "Graphics API: Vulkan or WebGpu")
endif()
set_property(CACHE NOORRHI_BACKEND PROPERTY STRINGS Vulkan WebGpu)
if(NOT NOORRHI_BACKEND MATCHES "^(Vulkan|WebGpu)$")
    message(FATAL_ERROR "NOORRHI_BACKEND must be Vulkan or WebGpu, not ${NOORRHI_BACKEND}")
endif()

add_subdirectory(external)
add_subdirectory(src)
if(NOORRHI_BUILD_TOOLS)
    add_subdirectory(tools)
endif()
if(NOORRHI_BUILD_TESTS)
    enable_testing()
    add_subdirectory(tests)
endif()
